# Reproduce the Analytical_workflow.ipynb processing with the smICA TCP API.
#
# Start smICA in another terminal before running this file:
#   run_smICA --automation-server --automation-port 8765
#
# The directory name "tcscp" is preserved because it is the name used in the
# reference-data tree. It denotes TCSPC statistical filtering.

set -euo pipefail

PORT="${SMICA_AUTOMATION_PORT:-8765}"
TIMEOUT="${SMICA_AUTOMATION_TIMEOUT:-3600}"
SCRIPT_DIRECTORY="$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd)"
REPOSITORY_ROOT="$(cd -- "$SCRIPT_DIRECTORY/.." && pwd)"
DATA_ROOT="$REPOSITORY_ROOT/calc_tests/REFERENCE_DATA/TypeOfFiltering"
RAW_ROOT="$DATA_ROOT/raw"
GATING_ROOT="$DATA_ROOT/gating"
TCSPC_ROOT="$DATA_ROOT/tcscp"
STATE_FILE="$SCRIPT_DIRECTORY/workflow_state.txt"
REPORT_FILE="$SCRIPT_DIRECTORY/workflow_report.txt"
RESUME="${RESUME_WORKFLOW:-true}"
START_TIME="$(date +%s)"
STARTED_AT="$(date --iso-8601=seconds 2>/dev/null || date)"
COMPLETED_EXTRACTIONS=0
SKIPPED_EXTRACTIONS=0
COMPLETED_CALCULATIONS=0
SKIPPED_CALCULATIONS=0

touch "$STATE_FILE"

call() {
    local command="$1"
    local parameters="${2-}"
    if [[ -z "$parameters" ]]; then
        parameters='{}'
    fi
    smica-call --port "$PORT" --timeout "$TIMEOUT" "$command" "$parameters"
}

resume_has() {
    [[ "$RESUME" == true ]] && grep -Fqx -- "$1" "$STATE_FILE"
}

mark_done() {
    if ! grep -Fqx -- "$1" "$STATE_FILE"; then
        printf '%s\n' "$1" >> "$STATE_FILE"
    fi
}

begin_extraction() {
    local key="$1"
    if resume_has "$key"; then
        printf 'RESUME: skipping extraction %s\n' "$key"
        ((SKIPPED_EXTRACTIONS += 1))
        return 1
    fi
    return 0
}

finish_extraction() {
    mark_done "$1"
    ((COMPLETED_EXTRACTIONS += 1))
}

begin_calculation() {
    local key="$1"
    local output="$2"
    if resume_has "$key" && [[ -s "$output" ]]; then
        printf 'RESUME: skipping calculation %s\n' "$key"
        ((SKIPPED_CALCULATIONS += 1))
        return 1
    fi
    return 0
}

finish_calculation() {
    mark_done "$1"
    ((COMPLETED_CALCULATIONS += 1))
}

write_report() {
    local exit_code=$?
    local finished_time elapsed status
    trap - EXIT
    finished_time="$(date +%s)"
    elapsed=$((finished_time - START_TIME))
    if ((exit_code == 0)); then
        status=COMPLETED
    else
        status="FAILED (exit code $exit_code)"
    fi
    {
        echo "smICA reference workflow report"
        echo "Status: $status"
        echo "Started: $STARTED_AT"
        echo "Finished: $(date --iso-8601=seconds 2>/dev/null || date)"
        printf 'Elapsed: %02d:%02d:%02d (%d seconds)\n' \
            "$((elapsed / 3600))" "$(((elapsed % 3600) / 60))" "$((elapsed % 60))" "$elapsed"
        echo "Completed extraction batches: $COMPLETED_EXTRACTIONS"
        echo "Skipped extraction batches: $SKIPPED_EXTRACTIONS"
        echo "Completed calculation batches: $COMPLETED_CALCULATIONS"
        echo "Skipped calculation batches: $SKIPPED_CALCULATIONS"
        echo "Resume state: $STATE_FILE"
        echo "Host: $(hostname 2>/dev/null || echo unavailable)"
        echo "Operating system: $(uname -s 2>/dev/null || echo unavailable)"
        echo "Kernel: $(uname -r 2>/dev/null || echo unavailable)"
        echo "Architecture: $(uname -m 2>/dev/null || echo unavailable)"
        echo "CPU: $(lscpu 2>/dev/null | awk -F: '/Model name/{sub(/^[[:space:]]+/, "", $2); print $2; exit}' || echo unavailable)"
        echo "Logical CPUs: $(getconf _NPROCESSORS_ONLN 2>/dev/null || echo unavailable)"
        echo "Memory: $(awk '/MemTotal/{printf "%.2f GiB", $2 / 1024 / 1024}' /proc/meminfo 2>/dev/null || echo unavailable)"
        echo "Python: $(${PYTHON_EXECUTABLE:-python} --version 2>&1 || echo unavailable)"
    } | tee "$REPORT_FILE"
    exit "$exit_code"
}
trap write_report EXIT

load_extract_directory() {
    local directory="$1"
    local skip_rows="$2"
    shift 2
    call extract.load_directory "{\"directory\":\"$directory\"}"
    call extract.set_rows_to_skip "{\"enabled\":$skip_rows,\"rows\":15}"

    local channel
    for channel in "$@"; do
        call extract.set_statistical_filter "{\"channel\":$channel,\"enabled\":false}"
    done
}

extract_raw_standard() {
    local directory="$1" key="extraction:raw:${1#$RAW_ROOT/}"
    begin_extraction "$key" || return 0
    load_extract_directory "$directory" false 1
    call extract.apply_all
    finish_extraction "$key"
}

extract_raw_pie() {
    local directory="$1"
    local key="extraction:raw:${1#$RAW_ROOT/}"
    local channel_1="$2"
    local channel_2="$3"
    begin_extraction "$key" || return 0
    local channels=()
    if [[ "$channel_1" == true ]]; then
        channels+=(1)
    fi
    if [[ "$channel_2" == true ]]; then
        channels+=(2)
    fi
    load_extract_directory "$directory" true "${channels[@]}"

    # set_gate requires both limits. The unchanged partner limit is repeated.
    # In the no-filtering workflow only: Channel 1 Time>0, Channel 2 Time<50.
    if [[ "$channel_1" == true ]]; then
        call extract.set_gate '{"channel":1,"lower_ns":0,"upper_ns":50}'
    fi
    if [[ "$channel_2" == true ]]; then
        call extract.set_gate '{"channel":2,"lower_ns":0,"upper_ns":50}'
    fi
    call extract.apply_all
    finish_extraction "$key"
}

extract_gating_pie() {
    local directory="$1"
    local key="extraction:gating:${1#$GATING_ROOT/}"
    begin_extraction "$key" || return 0
    shift
    load_extract_directory "$directory" true "$@"
    # Loading a PIE directory initializes the standard PIE gates used by the notebook.
    call extract.apply_all
    finish_extraction "$key"
}

extract_tcspc() {
    local directory="$1"
    local key="extraction:tcscp:${1#$TCSPC_ROOT/}"
    local skip_rows="$2"
    shift 2
    local channels=("$@")
    begin_extraction "$key" || return 0
    load_extract_directory "$directory" "$skip_rows" "${channels[@]}"

    if [[ "$(find "$directory" -maxdepth 1 -type f -name '*.ptu' -printf '.' | wc -c)" -eq 0 ]]; then
        printf 'No PTU files found in: %s\n' "$directory" >&2
        return 1
    fi

    # The first sorted PTU defines the gates and statistical-filter routine.
    # The native batch callback recalculates file-specific weights for all PTUs.
    local channel
    for channel in "${channels[@]}"; do
        call extract.set_statistical_filter "{\"channel\":$channel,\"enabled\":true}"
        call extract.open_filter_setup "{\"channel\":$channel}"
        call extract.set_background_range "{\"channel\":$channel,\"enabled\":true}"
        call extract.calculate_filters "{\"channel\":$channel}"
        call extract.accept_filters "{\"channel\":$channel}"
    done
    call extract.apply_all
    finish_extraction "$key"
}

load_calibration() {
    call phot2conc.load_calibration "{\"path\":\"$1\"}"
}

load_phot_directory() {
    local directory="$1"
    call phot2conc.load_directory "{\"directory\":\"$directory\"}"
    call phot2conc.set_roi_source '{"source":"none"}'
}

export_results() {
    local directory="$1"
    call phot2conc.export_results "{\"path\":\"$directory/results.csv\"}"
}

calculate_standard() {
    local directory="$1"
    local calibration="$2"
    local output="$directory/results.csv"
    local key="calculation:${directory#$DATA_ROOT/}/results.csv"
    begin_calculation "$key" "$output" || return 0
    load_phot_directory "$directory"
    load_calibration "$calibration"
    call phot2conc.calculate_all
    export_results "$directory"
    finish_calculation "$key"
}

calculate_raw_mrna() {
    local directory="$RAW_ROOT/pie_mrna_256x256_200_frames/shared_pie_calibration"
    local output="$directory/results.csv" key="calculation:raw/pie_mrna_256x256_200_frames/shared_pie_calibration/results.csv"
    begin_calculation "$key" "$output" || return 0
    load_phot_directory "$directory"
    load_calibration "$directory/calib.json"
    call phot2conc.set_roi_source '{"source":"auto"}'

    call phot2conc.select_file '{"filename":"cell3-1240-cyt1-mRNA-None_200nm_4.36us_4uW.ptu"}'
    call phot2conc.set_auto_roi '{"channel":1,"cell_threshold":0.1}'
    call phot2conc.calculate
    call phot2conc.add_result

    call phot2conc.select_file '{"filename":"cell3-1910-cyt1-mRNA-None_200nm_4.36us_4uW.ptu"}'
    call phot2conc.set_auto_roi '{"channel":1,"cell_threshold":0.05}'
    call phot2conc.calculate
    call phot2conc.add_result
    export_results "$directory"
    finish_calculation "$key"
}
